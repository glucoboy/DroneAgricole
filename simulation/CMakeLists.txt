cmake_minimum_required(VERSION 3.22)
project(CartographeSimulation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 REQUIRED COMPONENTS Core)
include(CTest)
add_executable(cartographe-evidence contracts/Evidence.cpp)
target_link_libraries(cartographe-evidence PRIVATE Qt6::Core)
target_compile_definitions(cartographe-evidence PRIVATE WORKSPACE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/..")
add_test(NAME cartographe_t0_contracts COMMAND cartographe-evidence self-test)
add_test(NAME cartographe_sitl_t0_contracts COMMAND cartographe-evidence self-test-sitl)
set_tests_properties(cartographe_t0_contracts PROPERTIES TIMEOUT 30)
set_tests_properties(cartographe_sitl_t0_contracts PROPERTIES TIMEOUT 30)
add_executable(test_offboard_safety tests/TestOffboardSafety.cpp)
target_include_directories(test_offboard_safety PRIVATE ../software/src/drone_agricole_px4_bridge/include)
add_test(NAME cartographe_offboard_safety COMMAND test_offboard_safety)
set_tests_properties(cartographe_offboard_safety PROPERTIES TIMEOUT 10)
add_executable(cartographe-model gazebo/ModelBuilder.cpp)
target_link_libraries(cartographe-model PRIVATE Qt6::Core)
add_test(NAME cartographe_model_generation COMMAND cartographe-model ${CMAKE_CURRENT_SOURCE_DIR}/..)
add_test(NAME cartographe_model_reject_unknown_mass COMMAND cartographe-model ${CMAKE_CURRENT_SOURCE_DIR}/.. --physical)
set_tests_properties(cartographe_model_generation cartographe_model_reject_unknown_mass PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
set_tests_properties(cartographe_model_reject_unknown_mass PROPERTIES WILL_FAIL TRUE)
add_executable(test_mass_properties mass/TestMass.cpp)
add_test(NAME cartographe_mass_properties COMMAND test_mass_properties)
set_tests_properties(cartographe_mass_properties PROPERTIES TIMEOUT 10)
add_executable(cartographe-bom-mass mass/BomMass.cpp)
target_link_libraries(cartographe-bom-mass PRIVATE Qt6::Core)
add_test(NAME cartographe_bom_mass_contract COMMAND cartographe-bom-mass --self-test)
set_tests_properties(cartographe_bom_mass_contract PROPERTIES TIMEOUT 10)
add_library(cartographe_energy STATIC energy/EnergyModel.cpp)
target_include_directories(cartographe_energy PUBLIC energy)
add_executable(cartographe-energy energy/energy_main.cpp)
target_link_libraries(cartographe-energy PRIVATE cartographe_energy Qt6::Core)
add_executable(test_cartographe_energy energy/TestEnergy.cpp)
target_link_libraries(test_cartographe_energy PRIVATE cartographe_energy)
add_test(NAME cartographe_energy_model COMMAND test_cartographe_energy)
set_tests_properties(cartographe_energy_model PROPERTIES TIMEOUT 30)
