cmake_minimum_required(VERSION 3.21)
project(convergence_manager VERSION 1.0.0 LANGUAGES CXX)

include(CTest)
find_package(Qt6 REQUIRED COMPONENTS Core)

add_library(convergence STATIC src/AdmissionScope.cpp src/Campaign.cpp src/Files.cpp src/Selection.cpp)
target_compile_features(convergence PUBLIC cxx_std_17)
set_target_properties(convergence PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(convergence PUBLIC src)
target_link_libraries(convergence PUBLIC Qt6::Core)

add_executable(convergence_core src/Main.cpp src/Admission.cpp src/ProcessBackend.cpp
    ../architecture_decision_engine/src/ResearchAdapter.cpp ../architecture_decision_engine/src/Evidence.cpp)
target_include_directories(convergence_core PRIVATE ../architecture_decision_engine/src)
target_link_libraries(convergence_core PRIVATE convergence)

if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    option(L5_V21_GO3A_TEST "Rejouer la porte V2.1 avec les archives GO3A/GO3B et leurs 180 dépendances" OFF)
    if(L5_V21_GO3A_TEST)
        add_test(NAME v21_gate COMMAND "${Python3_EXECUTABLE}" -B
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_admission.py" "$<TARGET_FILE:convergence_core>")
        set_tests_properties(v21_gate PROPERTIES TIMEOUT 180 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
    endif()
    foreach(case Campaign Selection)
        add_executable(test_l5_${case} tests/Test${case}.cpp)
        target_link_libraries(test_l5_${case} PRIVATE convergence)
        add_test(NAME l5_${case} COMMAND test_l5_${case} "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/campaign.json")
        set_tests_properties(l5_${case} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
    endforeach()
    add_executable(test_l5_v3_admission_entry tests/TestV3AdmissionEntry.cpp)
    target_link_libraries(test_l5_v3_admission_entry PRIVATE convergence)
    add_dependencies(test_l5_v3_admission_entry convergence_core)
    add_test(NAME l5_v3_admission_entry COMMAND test_l5_v3_admission_entry "$<TARGET_FILE:convergence_core>"
        "${CMAKE_CURRENT_BINARY_DIR}/trace-entry.json")
    set_tests_properties(l5_v3_admission_entry PROPERTIES TIMEOUT 90 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
    add_executable(test_l5_AdmissionScope tests/TestAdmissionScope.cpp)
    target_link_libraries(test_l5_AdmissionScope PRIVATE convergence)
    add_test(NAME l5_v3_admission_policy COMMAND test_l5_AdmissionScope
        "${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_CURRENT_BINARY_DIR}/trace-policy.json")
    set_tests_properties(l5_v3_admission_policy PROPERTIES TIMEOUT 90 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
    set(L5_L1_EXECUTABLE "" CACHE FILEPATH "Binaire de recherche L1")
    set(L5_L3_EXECUTABLE "" CACHE FILEPATH "Binaire d'exploration L3")
    set(L5_L4_EXECUTABLE "" CACHE FILEPATH "Binaire de validation L4")
    set(L5_FREECAD_PYTHON "" CACHE FILEPATH "Interpréteur FreeCAD pour la réception native")
    set(L5_T0_VALIDATOR "" CACHE FILEPATH "Validateur indépendant des preuves T0")
    if(L5_L1_EXECUTABLE AND L5_L3_EXECUTABLE AND L5_L4_EXECUTABLE)
        set(integration_args --core "$<TARGET_FILE:convergence_core>"
            --l1 "${L5_L1_EXECUTABLE}" --l3 "${L5_L3_EXECUTABLE}" --l4 "${L5_L4_EXECUTABLE}")
        if(L5_FREECAD_PYTHON)
            list(APPEND integration_args --freecad "${L5_FREECAD_PYTHON}")
        endif()
        if(L5_T0_VALIDATOR)
            list(APPEND integration_args --evidence "${L5_T0_VALIDATOR}")
        endif()
        add_test(NAME l5_cli COMMAND "${Python3_EXECUTABLE}" -B
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_campaign.py" ${integration_args})
        set_tests_properties(l5_cli PROPERTIES TIMEOUT 360 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
    endif()
endif()
