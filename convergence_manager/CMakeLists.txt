cmake_minimum_required(VERSION 3.21)
project(convergence_manager VERSION 1.0.0 LANGUAGES CXX)

include(CTest)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(OpenSSL 3 QUIET COMPONENTS Crypto)
if(NOT TARGET electromechanical_graph)
    add_subdirectory(../electromechanical_graph graph)
endif()

add_library(convergence STATIC src/AdmissionScope.cpp src/Campaign.cpp src/Files.cpp src/Selection.cpp src/V3Domain.cpp src/V3Search.cpp)
target_compile_features(convergence PUBLIC cxx_std_17)
set_target_properties(convergence PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(convergence PUBLIC src)
target_link_libraries(convergence PUBLIC Qt6::Core)

add_executable(convergence_core src/Main.cpp src/Admission.cpp src/ProcessBackend.cpp src/V3Transaction.cpp src/V3Campaign.cpp
    src/AuthorityReview.cpp src/MaterialEvidence.cpp src/A6Transaction.cpp src/A6Journal.cpp
    ../architecture_decision_engine/src/ResearchAdapter.cpp ../architecture_decision_engine/src/Evidence.cpp)
target_include_directories(convergence_core PRIVATE ../architecture_decision_engine/src)
target_link_libraries(convergence_core PRIVATE convergence electromechanical_graph)
set(L5_A5_PHYSICAL_TRUST_SHA256 "" CACHE STRING "SHA-256 provisionné de la racine de confiance matérielle A.5")
if(L5_A5_PHYSICAL_TRUST_SHA256)
    string(LENGTH "${L5_A5_PHYSICAL_TRUST_SHA256}" a5_trust_hash_length)
    if(NOT a5_trust_hash_length EQUAL 64 OR NOT L5_A5_PHYSICAL_TRUST_SHA256 MATCHES "^[0-9a-f]+$")
        message(FATAL_ERROR "Empreinte de confiance A.5 invalide")
    endif()
endif()
target_compile_definitions(convergence_core PRIVATE V3_A5_PHYSICAL_TRUST_SHA256="${L5_A5_PHYSICAL_TRUST_SHA256}")
# Les outils A.6 viennent du déploiement du binaire de banc, sans option de requête.
set(L5_FREECAD_PYTHON "" CACHE FILEPATH "Interpréteur FreeCAD pour la réception native")
set(L5_T0_VALIDATOR "" CACHE FILEPATH "Validateur indépendant des preuves T0")
file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/.." a6_checkout)
target_compile_definitions(convergence_core PRIVATE
    V3_A6_CHECKOUT="${a6_checkout}" V3_A6_FREECAD="${L5_FREECAD_PYTHON}" V3_A6_T0="${L5_T0_VALIDATOR}")
option(L5_A6_FAULT_INJECTION "Activer les interruptions bornées du banc A.6" OFF)
mark_as_advanced(L5_A6_FAULT_INJECTION)
if(L5_A6_FAULT_INJECTION)
    if(NOT BUILD_TESTING OR L5_A5_PHYSICAL_TRUST_SHA256)
        message(FATAL_ERROR "Les interruptions A.6 exigent un build de test sans racine physique")
    endif()
    target_compile_definitions(convergence_core PRIVATE V3_A6_FAULT_INJECTION)
endif()

if(TARGET OpenSSL::Crypto)
    target_compile_definitions(convergence_core PRIVATE V3_A5_CRYPTO)
    target_link_libraries(convergence_core PRIVATE OpenSSL::Crypto)
endif()

add_executable(cartographe_material_evidence src/MaterialEvidence.cpp)
target_compile_definitions(cartographe_material_evidence PRIVATE V3_A4_IMPORT_EXECUTABLE)
target_link_libraries(cartographe_material_evidence PRIVATE convergence)

set(a7_matrix "${CMAKE_CURRENT_SOURCE_DIR}/../docs/verification/v3/a7/evidence_matrix.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${a7_matrix}")
file(SHA256 "${a7_matrix}" a7_matrix_sha256)
add_executable(cartographe_reception src/A7Reception.cpp src/MaterialEvidence.cpp)
set_target_properties(cartographe_reception PROPERTIES CXX_EXTENSIONS OFF)
target_compile_definitions(cartographe_reception PRIVATE V3_A7_MATRIX_SHA256="${a7_matrix_sha256}")
target_link_libraries(cartographe_reception PRIVATE convergence)

if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME l5_v3_a7_reception COMMAND "${Python3_EXECUTABLE}" -B
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_v3_a7_reception.py"
        --tool "$<TARGET_FILE:cartographe_reception>")
    add_test(NAME l5_v3_a4_material_evidence COMMAND "${Python3_EXECUTABLE}" -B
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_v3_a4_evidence.py"
        --validator "$<TARGET_FILE:cartographe_material_evidence>")
    if(TARGET OpenSSL::Crypto)
        find_program(L5_A5_OPENSSL_EXECUTABLE openssl HINTS "${OPENSSL_ROOT_DIR}/bin")
        if(L5_A5_OPENSSL_EXECUTABLE AND NOT L5_A5_PHYSICAL_TRUST_SHA256)
            add_test(NAME l5_v3_a5_authority COMMAND "${Python3_EXECUTABLE}" -B
                "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_v3_a5_authority.py"
                --core "$<TARGET_FILE:convergence_core>" --openssl "${L5_A5_OPENSSL_EXECUTABLE}")
            set_tests_properties(l5_v3_a5_authority PROPERTIES TIMEOUT 120)
        endif()
    endif()
    option(L5_V21_GO3A_TEST "Rejouer la porte V2.1 avec les archives GO3A/GO3B et leurs 180 dépendances" OFF)
    if(L5_V21_GO3A_TEST)
        add_test(NAME v21_gate COMMAND "${Python3_EXECUTABLE}" -B
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_admission.py" "$<TARGET_FILE:convergence_core>")
        set_tests_properties(v21_gate PROPERTIES TIMEOUT 180 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
    endif()
    foreach(case Campaign Selection)
        add_executable(test_l5_${case} tests/Test${case}.cpp)
        target_link_libraries(test_l5_${case} PRIVATE convergence)
        add_test(NAME l5_${case} COMMAND test_l5_${case} "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/campaign.json")
        set_tests_properties(l5_${case} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
    endforeach()
    add_executable(test_l5_v3_admission_entry tests/TestV3AdmissionEntry.cpp)
    target_link_libraries(test_l5_v3_admission_entry PRIVATE convergence)
    add_dependencies(test_l5_v3_admission_entry convergence_core)
    add_test(NAME l5_v3_admission_entry COMMAND test_l5_v3_admission_entry "$<TARGET_FILE:convergence_core>"
        "${CMAKE_CURRENT_BINARY_DIR}/trace-entry.json")
    set_tests_properties(l5_v3_admission_entry PROPERTIES TIMEOUT 90 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
    add_executable(test_l5_AdmissionScope tests/TestAdmissionScope.cpp)
    target_link_libraries(test_l5_AdmissionScope PRIVATE convergence)
    add_test(NAME l5_v3_admission_policy COMMAND test_l5_AdmissionScope
        "${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_CURRENT_BINARY_DIR}/trace-policy.json")
    set_tests_properties(l5_v3_admission_policy PROPERTIES TIMEOUT 90 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
    add_executable(test_l5_V3Search tests/TestV3Search.cpp)
    target_link_libraries(test_l5_V3Search PRIVATE convergence)
    set(L5_L1_EXECUTABLE "" CACHE FILEPATH "Binaire de recherche L1")
    set(L5_L3_EXECUTABLE "" CACHE FILEPATH "Binaire d'exploration L3")
    set(L5_L4_EXECUTABLE "" CACHE FILEPATH "Binaire de validation L4")
    if(L5_FREECAD_PYTHON)
        foreach(worker mutate verify)
            add_test(NAME l5_v3_native_${worker} COMMAND "${Python3_EXECUTABLE}" -B
                "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_v3_${worker}.py" --freecad "${L5_FREECAD_PYTHON}" -v)
            set_tests_properties(l5_v3_native_${worker} PROPERTIES TIMEOUT 180
                WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
        endforeach()
    endif()
    if(L5_L1_EXECUTABLE AND L5_L3_EXECUTABLE AND L5_L4_EXECUTABLE)
        set(integration_args --core "$<TARGET_FILE:convergence_core>"
            --l1 "${L5_L1_EXECUTABLE}" --l3 "${L5_L3_EXECUTABLE}" --l4 "${L5_L4_EXECUTABLE}")
        if(L5_FREECAD_PYTHON)
            list(APPEND integration_args --freecad "${L5_FREECAD_PYTHON}")
        endif()
        if(L5_T0_VALIDATOR)
            list(APPEND integration_args --evidence "${L5_T0_VALIDATOR}")
        endif()
        add_test(NAME l5_cli COMMAND "${Python3_EXECUTABLE}" -B
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_campaign.py" ${integration_args})
        set_tests_properties(l5_cli PROPERTIES TIMEOUT 360 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
    endif()
endif()
