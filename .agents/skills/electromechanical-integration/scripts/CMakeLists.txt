cmake_minimum_required(VERSION 3.20)
project(electromechanical_integration LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core)
add_library(harness_planner_core STATIC harness_planner.cpp)
target_compile_definitions(harness_planner_core PRIVATE HARNESS_PLANNER_LIBRARY)
target_link_libraries(harness_planner_core PUBLIC Qt6::Core)
add_executable(harness-planner harness_planner.cpp)
target_link_libraries(harness-planner PRIVATE Qt6::Core)
add_executable(harness-planner-tests tests_planner.cpp)
target_link_libraries(harness-planner-tests PRIVATE harness_planner_core)
enable_testing()
add_test(NAME planner_properties COMMAND harness-planner-tests)
